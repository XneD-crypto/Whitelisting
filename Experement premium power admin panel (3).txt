local function getRealHWID()
    if gethwid then
        local ok, id = pcall(gethwid)
        if ok and id and id ~= "" then return tostring(id) end
    end
    if delta then
        if delta.gethwid then
            local ok, id = pcall(delta.gethwid)
            if ok and id then return tostring(id) end
        end
        if delta.HWID then return tostring(delta.HWID) end
    end
    if secure_load then
        local ok, id = pcall(secure_load, "return gethwid()")
        if ok and id then return tostring(id) end
    end
    return nil
end
